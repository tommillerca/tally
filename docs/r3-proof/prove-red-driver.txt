const proof='/tmp/r3-red-output'; fs.mkdirSync(proof,{recursive:true});
function run(name,file,want){
 const r=spawnSync(process.execPath,[path.join(temp,'tests',file)],{encoding:'utf8',timeout:120000});
 fs.writeFileSync(path.join(proof,name+'.txt'),(r.stdout||'')+(r.stderr||''));
 fs.writeFileSync(path.join(proof,name+'.exit'),String(r.status));
 console.log(name, r.status);assert.equal(r.status,want);
}

}
