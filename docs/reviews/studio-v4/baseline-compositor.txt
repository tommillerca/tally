// Project at the social boundary, before the compositor ever sees a snapshot.
export function studioCrewAppearance(friends) {
  return friends.map(f => ({ label: String(f.alias || f.name || f.handle || 'Crew friend'),
    outfit: Object.fromEntries(BH_SLOTS.filter(s => !['BG', 'C'].includes(s.code))
      .map(s => [s.code, f.profile?.outfit?.[s.code]])
      .filter(([slot, id]) => typeof id === 'string' && BH_BY_ID[id]?.slot === slot)) }));
}

// Coordinates are centres in export pixels. Array order is back to front.
export function studioStickerBox(ink, sticker) {
  const requestedScale = sticker.size / Math.max(ink.width, ink.height);
  let scale = requestedScale;
  const angle = (sticker.rotation || 0) * Math.PI / 180;
  const c = Math.abs(Math.cos(angle)), s = Math.abs(Math.sin(angle));
  // A 650px square at 45 degrees plus the 12px rim and hard shadow is
  // 952px wide. Uniformly cap only that geometric edge case to fit 950px.
  scale = Math.min(scale,
    (STUDIO_SAFE.right - STUDIO_SAFE.left - STUDIO_OUTLINE * 2 - STUDIO_SHADOW.x) / (ink.width * c + ink.height * s),
    (STUDIO_SAFE.bottom - STUDIO_SAFE.top - STUDIO_OUTLINE * 2 - STUDIO_SHADOW.y) / (ink.width * s + ink.height * c));
  const fittedSize = scale < requestedScale ? scale * Math.max(ink.width, ink.height) : sticker.size;
  const artWidth = Math.ceil((ink.width * c + ink.height * s) * scale - 1e-9);
  const artHeight = Math.ceil((ink.width * s + ink.height * c) * scale - 1e-9);
  const width = artWidth + STUDIO_OUTLINE * 2 + STUDIO_SHADOW.x;
  const height = artHeight + STUDIO_OUTLINE * 2 + STUDIO_SHADOW.y;
  return { x: Math.round(Math.max(STUDIO_SAFE.left, Math.min(STUDIO_SAFE.right - width, sticker.x - width / 2))),
    y: Math.round(Math.max(STUDIO_SAFE.top, Math.min(STUDIO_SAFE.bottom - height, sticker.y - height / 2))), width, height,
    artWidth, artHeight, fittedSize };
}

// Inverse-map output pixel centres directly to source pixels. Canvas transforms
// can antialias rotated edges even with smoothing disabled. Copying RGBA avoids
// that backend-dependent interpolation, including at arbitrary twist angles.
export function studioRasterSticker(figure, sticker, runtime) {
  const box = studioStickerBox(figure.ink, sticker);
  const key = [sticker.size, sticker.rotation || 0, sticker.flip].join(':');
  if (figure.raster?.key === key) return { ...figure.raster, box };
  const { width, height, artWidth, artHeight } = box, pad = STUDIO_OUTLINE;
  const art = runtime.createCanvas(width, height), ac = art.getContext('2d');
  const pixels = ac.createImageData(width, height);
  const src = figure.pixels ||= figure.canvas.getContext('2d').getImageData(0, 0, figure.canvas.width, figure.canvas.height).data;
  const b = figure.ink, scale = box.fittedSize / Math.max(b.width, b.height);
  const angle = (sticker.rotation || 0) * Math.PI / 180, co = Math.cos(angle), si = Math.sin(angle);
  for (let y = 0; y < artHeight; y++) for (let x = 0; x < artWidth; x++) {
    const dx = x + .5 - artWidth / 2, dy = y + .5 - artHeight / 2;
    const sx = Math.floor(b.x + b.width / 2 + (dx * co + dy * si) / scale * (sticker.flip ? -1 : 1));
    const sy = Math.floor(b.y + b.height / 2 + (-dx * si + dy * co) / scale);
    if (sx < b.x || sy < b.y || sx >= b.x + b.width || sy >= b.y + b.height) continue;
    const from = (sy * figure.canvas.width + sx) * 4, to = ((y + pad) * width + x + pad) * 4;
    pixels.data.set(src.subarray(from, from + 4), to);
  }
  ac.putImageData(pixels, 0, 0);
  // Exact integer disk dilation of nonzero alpha. Sliding row windows keep it
  // O(radius * pixels), rather than painting one disk per opaque pixel.
  const mask = new Uint8Array(width * height), dilated = new Uint8Array(mask.length);
  for (let i = 0; i < mask.length; i++) mask[i] = pixels.data[i * 4 + 3] ? 1 : 0;
  for (let dy = -pad; dy <= pad; dy++) {
    const radius = Math.floor(Math.sqrt(pad * pad - dy * dy));
    for (let y = Math.max(0, -dy); y < Math.min(height, height - dy); y++) {
      const row = y * width, out = (y + dy) * width;
      let count = 0;
      for (let x = 0; x <= radius && x < width; x++) count += mask[row + x];
      for (let x = 0; x < width; x++) {
        if (count) dilated[out + x] = 1;
        if (x - radius >= 0) count -= mask[row + x - radius];
        if (x + radius + 1 < width) count += mask[row + x + radius + 1];
      }
    }
  }
  const decoration = runtime.createCanvas(width, height), dc = decoration.getContext('2d');
  const border = dc.createImageData(width, height);
  const put = i => { border.data.set([42, 45, 40, 255], i * 4); };
  for (let y = 0; y < height; y++) for (let x = 0; x < width; x++) if (dilated[y * width + x]) {
    put(y * width + x);
    if (x + STUDIO_SHADOW.x < width && y + STUDIO_SHADOW.y < height)
      put((y + STUDIO_SHADOW.y) * width + x + STUDIO_SHADOW.x);
  }
  dc.putImageData(border, 0, 0);
  figure.raster = { key, art, decoration, mask, dilated };
  return { ...figure.raster, box };
}

const fail = message => { throw new Error(message); };
function keys(value, allowed, label) {
  if (!value || typeof value !== 'object' || Array.isArray(value)) fail(`Invalid ${label}.`);
  for (const key of Object.keys(value)) if (!allowed.includes(key)) fail(`Unsupported ${label} field: ${key}.`);
}
function itemFor(id, slot) {
  const item = BH_BY_ID[id];
  if (!item || item.slot !== slot) fail(`Unknown ${slot} art: ${id}.`);
  return item;
}

export function studioPlan(look, options = {}) {
  keys(look, ['outfit', 'pet', 'friendCode'], 'look');
  keys(look.outfit, BH_SLOTS.map(s => s.code), 'outfit');
  keys(options, Object.keys(STUDIO_DEFAULTS), 'option');
  const opts = { ...STUDIO_DEFAULTS, ...options };
  for (const flag of ['includePet', 'includeFriendCode']) if (typeof opts[flag] !== 'boolean') fail(`Invalid ${flag}.`);
  if (!Object.hasOwn(STUDIO_CAPTIONS, opts.caption)) fail('Choose a Studio caption. Free text is unavailable.');
  if (opts.frame !== null) fail('Frames await art review.');
  if (!STUDIO_POSITIONS.includes(opts.bubblePosition)) fail('Choose a bubble position.');
  if (!STUDIO_MARK_POSITIONS.includes(opts.markPosition)) fail('Choose a mark position.');
  if (!Array.isArray(opts.stickers) || opts.stickers.length > 12) fail('Choose up to 12 stickers.');
  const stickers = opts.stickers.map(s => {
    keys(s, ['kind', 'id', 'outfit', 'x', 'y', 'size', 'flip', 'rotation'], 'sticker');
    if (!Number.isFinite(s.rotation ?? 0)) fail('Invalid sticker rotation.');
    if (![s.x, s.y, s.size].every(Number.isFinite) || s.size < STUDIO_STICKER_MIN || s.size > STUDIO_STICKER_MAX || typeof s.flip !== 'boolean') fail('Invalid sticker transform.');
    if (s.kind === 'text') {
      if (!Object.hasOwn(STUDIO_TEXT_STICKERS, s.id) || s.outfit) fail('Unknown text sticker.');
      return { ...s, text: STUDIO_TEXT_STICKERS[s.id] };
    }
    if (!['monster', 'crew'].includes(s.kind)) fail('Unknown sticker kind.');
    if (s.kind === 'monster') {
      if (!Object.hasOwn(STUDIO_CREATURES, s.id) || s.outfit) fail('Unknown monster sticker.');
      return { ...s, layers: [{ src: STUDIO_CREATURES[s.id], group: 'creature' }] };
    }
    if (s.kind === 'crew' && s.id !== undefined) fail('Crew stickers carry appearance only.');
    const outfit = s.outfit;
    const layers = studioPlan({ outfit }, { includePet: false }).layers.filter(l => l.group === 'body');
    // No tint, filter or recolour is applied to placed Cam art.
    return { ...s, layers: layers.map(l => ({ ...l, tints: [] })) };
  });
  const layers = [];
  if (opts.backdrop !== null) layers.push({ slot: 'BG', group: 'backdrop', z: 0, src: bhAsset(itemFor(opts.backdrop, 'BG')) });
  for (const slot of [...BH_SLOTS].sort((a, b) => a.z - b.z)) {
    if (slot.code === 'BG' || slot.code === 'C') continue;
    const id = look.outfit[slot.code] ?? slot.default;
    if (!id) continue;
    const item = itemFor(id, slot.code);
    if (slot.code === 'E' && visorHidesEyes(look.outfit)) continue;
    layers.push({ slot: slot.code, group: 'body', z: slot.z, src: bhAsset(item),
      tints: footballTints(item), mask: slot.code === 'E' ? visorClipMask(look.outfit) : null });
  }
  if (opts.includePet && look.outfit.C && look.pet?.id !== look.outfit.C) fail('The equipped pet needs its matching instance appearance.');
  if (opts.includePet && look.pet) {
    const pet = look.pet;
    keys(pet, ['id', 'shiny', 'morph', 'wear'], 'pet');
    const item = itemFor(pet.id, 'C');
    if (typeof pet.shiny !== 'boolean') fail('Pet appearance requires explicit instance shiny.');
    if (pet.morph != null && !isMorph(pet.morph)) fail('Unknown pet colour.');
    if (pet.wear) {
      keys(pet.wear, PET_SLOTS.map(s => s.code), 'pet clothing');
      for (const [slot, id] of Object.entries(pet.wear)) if (id) itemFor(id, slot);
    }
    const worn = petWornItems(pet.id, pet.wear);
    if (Object.values(pet.wear || {}).filter(Boolean).length !== worn.length) fail('Clothing does not fit this pet.');
    const src = pet.shiny && pet.id !== 'CX' ? `assets/bh/C/shiny/${pet.id}.png`
      : morphAsset(pet.id, pet.morph) || bhAsset(item);
    layers.push({ slot: 'C', group: 'pet', z: 5, src });
    for (const w of worn) layers.push({ slot: w.slot, group: 'pet', z: PET_SLOTS.find(s => s.code === w.slot).z,
      src: bhAsset(w), tints: footballTints(w) });
  }
  const text = [];
  if (opts.caption !== 'none') text.push({ id: 'caption', text: STUDIO_CAPTIONS[opts.caption] });
  if (opts.includeFriendCode) {
    if (!/^BONE-[A-Z0-9]{4}-[A-Z0-9]{4}$/.test(look.friendCode || '')) fail('Your friend code is unavailable.');
    text.push({ id: 'friend-code', text: look.friendCode, x: 90, baseline: 1520, size: 38 });
  }
  return { layers, text, stickers, bubblePosition: opts.bubblePosition, markPosition: opts.markPosition, frame: null };
}

export function assertStudioSafe(elements) {
  if (!elements.length) fail('No informational elements measured.');
  for (const e of elements) {
    if (![e.x, e.y, e.width, e.height].every(Number.isFinite) || e.width <= 0 || e.height <= 0
      || e.x < STUDIO_SAFE.left || e.y < STUDIO_SAFE.top
      || e.x + e.width > STUDIO_SAFE.right || e.y + e.height > STUDIO_SAFE.bottom) {
      fail(`Unsafe Studio information: ${e.id}.`);
    }
  }
}

// Match drawTrimmedArt's ink threshold, after stacking at shared registration.
export function studioInk(canvas) {
  const { width, height } = canvas;
  const data = canvas.getContext('2d').getImageData(0, 0, width, height).data;
  let left = width, top = height, right = -1, bottom = -1;
  for (let y = 0; y < height; y++) for (let x = 0; x < width; x++) {
    if (data[(y * width + x) * 4 + 3] <= 14) continue;
    left = Math.min(left, x); top = Math.min(top, y); right = Math.max(right, x); bottom = Math.max(bottom, y);
  }
  if (right < left) fail('Required Studio art is empty.');
  return { x: left, y: top, width: right - left + 1, height: bottom - top + 1 };
}

// Exhaustive integer-pixel search: overlap first, distance to the chosen preset
// second. Integral alpha occupancy measures artwork, never backdrop colour.
export function studioClearPosition(canvas, box, obstacles = [], shape = null) {
  const w = canvas.width, h = canvas.height, stride = w + 1;
  const pixels = canvas.getContext('2d').getImageData(0, 0, w, h).data;
  const sums = new Uint32Array((w + 1) * (h + 1));
  for (let y = 0; y < h; y++) {
    let row = 0;
    for (let x = 0; x < w; x++) {
      row += pixels[(y * w + x) * 4 + 3] > 14 ? 1 : 0;
      sums[(y + 1) * stride + x + 1] = sums[y * stride + x + 1] + row;
    }
  }
  const width = Math.ceil(box.width), height = Math.ceil(box.height);
  const runs = []; let area = width * height;
  if (shape) {
    const alpha = shape.canvas.getContext('2d').getImageData(0, 0, width, height).data;
    area = 0;
    for (let y = 0; y < height; y++) for (let x = 0; x < width;) {
      if (alpha[(y * width + x) * 4 + 3] <= 14) { x++; continue; }
      const start = x;
      while (x < width && alpha[(y * width + x) * 4 + 3] > 14) x++;
      runs.push([y, start, x]); area += x - start;
    }
  }
  let best, score = Infinity, distance = Infinity;
  for (let y = STUDIO_SAFE.top; y <= STUDIO_SAFE.bottom - height; y++) {
    for (let x = STUDIO_SAFE.left; x <= STUDIO_SAFE.right - width; x++) {
      if (obstacles.some(b => x < b.x + b.width && x + width > b.x && y < b.y + b.height && y + height > b.y)) continue;
      if (shape?.accept && !shape.accept(x, y)) continue;
      let n = sums[(y + height) * stride + x + width] - sums[y * stride + x + width]
        - sums[(y + height) * stride + x] + sums[y * stride + x];
      const d = (x - box.x) ** 2 + (y - box.y) ** 2;
      if (shape && n) {
        n = 0;
        for (const [dy, left, right] of runs) {
          const row = (y + dy) * stride;
          n += sums[row + stride + x + right] - sums[row + x + right]
            - sums[row + stride + x + left] + sums[row + x + left];
          if (n > score || (n === score && d >= distance)) break;
        }
      }
      if (n < score || (n === score && d < distance)) {
        score = n; distance = d; best = { ...box, x, y, width, height, overlapPixels: n,
          overlapPercent: 100 * n / area };
      }
    }
  }
  if (!best && !shape?.accept) fail('No safe Studio placement.');
  return best;
}

// Compare final composited RGB with the immediate backing RGB at source-mask
// samples. Opaque letter cores are graded; antialiased edges are reported too.
export function studioMarkContrast(before, after, mask) {
  const luminance = (p, i) => [0.2126, 0.7152, 0.0722].reduce((sum, k, c) => {
    const v = p[i + c] / 255;
    return sum + k * (v <= .04045 ? v / 12.92 : ((v + .055) / 1.055) ** 2.4);
  }, 0);
  let minimum = Infinity, edgeMinimum = Infinity, samples = 0;
  for (let i = 0; i < mask.length; i += 4) {
    if (!mask[i + 3]) continue;
    const a = luminance(before, i), b = luminance(after, i);
    const ratio = (Math.max(a, b) + .05) / (Math.min(a, b) + .05);
    edgeMinimum = Math.min(edgeMinimum, ratio);
    if (mask[i + 3] === 255) { minimum = Math.min(minimum, ratio); samples++; }
  }
  if (!samples) fail('No opaque Studio wordmark samples.');
  return { minimum, edgeMinimum, samples };
}
export function assertStudioMarkContrast(measurement) {
  if (!Number.isFinite(measurement.minimum) || measurement.minimum < 4.5)
    fail(`Studio wordmark contrast ${measurement.minimum.toFixed(2)}:1 below 4.5:1.`);
}

let fontReady;
export function studioBrowserRuntime() {
  return {
    createCanvas(width, height) { const c = document.createElement('canvas'); c.width = width; c.height = height; return c; },
    async loadImage(src) {
      const img = new Image(); img.src = src;
      let timer;
      try {
        await Promise.race([img.decode(), new Promise((_, reject) => {
          timer = setTimeout(() => { img.src = ''; reject(new Error(`Art timed out: ${src}`)); }, 15000);
        })]);
        if (!img.naturalWidth) fail(`Art unavailable: ${src}`);
        return img;
      } finally { clearTimeout(timer); }
    },
    async ready() {
      if (!fontReady) fontReady = (async () => {
        const font = new FontFace('StudioBangers', 'url(assets/fonts/bangers.woff2)');
        document.fonts.add(await font.load());
        const dialogue = new FontFace('StudioDialogue', 'url(assets/fonts/boldpixels.woff2)');
        document.fonts.add(await dialogue.load());
      })().catch(error => { fontReady = null; throw error; });
      await fontReady;
    },
    encode: canvas => new Promise((resolve, reject) => canvas.toBlob(blob => blob ? resolve(blob) : reject(new Error('Image encoding failed.')), 'image/png')),
  };
}

// The tray and placed art share exactly the same source stack and treatment.
export async function studioStickerFigure(sticker, runtime = studioBrowserRuntime(), load = src => runtime.loadImage(src)) {
  await runtime.ready();
  const s = studioPlan({ outfit: {} }, { includePet: false, stickers: [sticker] }).stickers[0];
  if (s.kind === 'text') {
    const c = runtime.createCanvas(800, 170), cc = c.getContext('2d');
    cc.font = '76px StudioBangers'; cc.textBaseline = 'middle';
    cc.fillStyle = '#F3EFE7'; cc.fillText(s.text, 12, 85);
    return { canvas: c, ink: studioInk(c) };
  }
  const images = [];
  for (const layer of s.layers) images.push(await load(layer.src));
  if (s.kind === 'monster') {
    const image = images[0], c = runtime.createCanvas(image.width, image.height);
    c.getContext('2d').drawImage(image, 0, 0);
    return { canvas: c, ink: studioInk(c) };
  }
  const size = Math.max(...images.map(i => Math.max(i.width, i.height)));
  const c = runtime.createCanvas(size, size), cc = c.getContext('2d'); cc.imageSmoothingEnabled = false;
  for (let i = 0; i < images.length; i++) {
    const layer = runtime.createCanvas(size, size), lc = layer.getContext('2d'); lc.imageSmoothingEnabled = false;
    lc.drawImage(images[i], 0, 0, size, size);
    if (s.layers[i].mask) { lc.globalCompositeOperation = 'destination-in'; lc.drawImage(await load(s.layers[i].mask), 0, 0, size, size); }
    cc.drawImage(layer, 0, 0);
  }
  return { canvas: c, ink: studioInk(c) };
}

export async function composeStudio(look, options = {}, runtime = studioBrowserRuntime()) {
  const plan = studioPlan(look, options);
  await runtime.ready();
  // Scoped cache is released after this render, and includes masks. No partial exports.
  const assets = new Map();
  const load = async src => {
    if (!assets.has(src)) assets.set(src, runtime.loadImage(src).catch(() => fail(`Download required art and retry: ${src}`)));
    return assets.get(src);
  };
  const required = [...new Set([...plan.layers, ...plan.stickers.flatMap(s => s.layers || []), { src: 'assets/brand/wordmark.png' }].flatMap(l => [l.src, l.mask, ...(l.tints || []).map(t => t.mask)]).filter(Boolean))];
  // Sequential decode bounds the cold-load memory spike from 2048px pet clothing.
  for (const src of required) await load(src);
  const canvas = runtime.createCanvas(1080, 1920);
  let ctx = canvas.getContext('2d');
  ctx.fillStyle = '#F3EFE7'; ctx.fillRect(0, 0, 1080, 1920);
  const bg = plan.layers.find(l => l.group === 'backdrop');
  if (bg) {
    const img = await load(bg.src), scale = Math.max(1080 / img.width, 1920 / img.height);
    ctx.drawImage(img, (1080 - img.width * scale) / 2, (1920 - img.height * scale) / 2, img.width * scale, img.height * scale);
  }
  const stack = async (group, input = plan.layers) => {
    const layers = input.filter(l => l.group === group);
    if (!layers.length) return null;
    const size = Math.max(...(await Promise.all(layers.map(l => load(l.src)))).map(i => Math.max(i.width, i.height)));
    const c = runtime.createCanvas(size, size), cctx = c.getContext('2d');
    cctx.imageSmoothingEnabled = false;
    for (const l of layers) {
      const layer = runtime.createCanvas(size, size), lc = layer.getContext('2d');
      lc.imageSmoothingEnabled = false;
      lc.drawImage(await load(l.src), 0, 0, size, size);
      if (l.mask) { lc.globalCompositeOperation = 'destination-in'; lc.drawImage(await load(l.mask), 0, 0, size, size); lc.globalCompositeOperation = 'source-over'; }
      // Existing catalogue colour overlays only, identical to the wardrobe's
      // multiply masks. Never generate a colourway or filter Cam's base art.
      for (const t of l.tints || []) {
        const mask = runtime.createCanvas(size, size), mc = mask.getContext('2d');
        mc.drawImage(await load(t.mask), 0, 0, size, size);
        mc.globalCompositeOperation = 'source-in'; mc.fillStyle = t.hex; mc.fillRect(0, 0, size, size);
        lc.globalCompositeOperation = 'multiply'; lc.drawImage(mask, 0, 0); lc.globalCompositeOperation = 'source-over';
      }
      cctx.drawImage(layer, 0, 0);
    }
    return { canvas: c, ink: studioInk(c) };
  };
  const body = await stack('body'), pet = await stack('pet');
  if (!body) fail('No Bonehead art.');
  const ground = 1460, bounds = [], information = [];
  const occupancy = runtime.createCanvas(1080, 1920), oc = occupancy.getContext('2d');
  oc.imageSmoothingEnabled = false;
  ctx.imageSmoothingEnabled = false; // nearest-neighbour, no interpolated art colours
  const paint = (figure, box, flip = false) => {
    const b = figure.ink;
    ctx.save(); ctx.translate(box.x + (flip ? box.width : 0), box.y); ctx.scale(flip ? -1 : 1, 1);
    ctx.drawImage(figure.canvas, b.x, b.y, b.width, b.height, 0, 0, box.width, box.height); ctx.restore();
  };
  const bodyScale = Math.min(850 / body.ink.width, 1120 / body.ink.height);
  const bodyBox = { id: 'body', x: 65, y: ground - body.ink.height * bodyScale,
    width: body.ink.width * bodyScale, height: body.ink.height * bodyScale };
  // Ground by the feet, not a sword or other accessory extending below them.
  const feet = await stack('body', plan.layers.filter(l => l.slot === (plan.layers.some(l => l.slot === 'FW') ? 'FW' : 'B')));
  const feetY = bodyBox.y + ((feet.ink.y + feet.ink.height) * body.canvas.width / feet.canvas.width - body.ink.y) * bodyScale;
  if (pet) {
    const scale = Math.min(350 / pet.ink.width, 460 / pet.ink.height);
    const box = { id: 'pet', x: 1010 - pet.ink.width * scale, y: feetY - pet.ink.height * scale,
      width: pet.ink.width * scale, height: pet.ink.height * scale };
    paint(pet, box); bounds.push(box);
    oc.drawImage(pet.canvas, pet.ink.x, pet.ink.y, pet.ink.width, pet.ink.height, box.x, box.y, box.width, box.height);
  }
  paint(body, bodyBox); bounds.push(bodyBox);
  oc.drawImage(body.canvas, body.ink.x, body.ink.y, body.ink.width, body.ink.height, bodyBox.x, bodyBox.y, bodyBox.width, bodyBox.height);

  const baseCanvas = runtime.createCanvas(1080, 1920);
  baseCanvas.getContext('2d').drawImage(canvas, 0, 0);
  const figures = [];
  for (const [index, sticker] of plan.stickers.entries()) {
    const figure = await studioStickerFigure(options.stickers[index], runtime, load);
    figures.push(figure);
    const raster = studioRasterSticker(figure, sticker, runtime), box = { id: `sticker-${index}`, ...raster.box };
    ctx.drawImage(raster.decoration, box.x, box.y);
    ctx.drawImage(raster.art, box.x, box.y); bounds.push(box);
  }
  // Record the information layer while painting the final export. Live gestures
  // reuse these decoded surfaces without loading, encoding or running placement.
  const foreground = runtime.createCanvas(1080, 1920), fc = foreground.getContext('2d'), main = ctx;
  ctx = new Proxy(main, {
    get(target, key) {
      const value = target[key];
      if (typeof value !== 'function') return value;
      if (['getImageData', 'measureText', 'getTransform'].includes(key)) return value.bind(target);
      return (...args) => { fc[key](...args); return value.apply(target, args); };
    },
    set(target, key, value) { target[key] = value; fc[key] = value; return true; },
  });
  ctx.imageSmoothingEnabled = false;
  // Port of talkBoxHtml's .talkbox + .hero-bubble at 3 export px per CSS px.
  // app.css: size 11, line-height 1.55, tracking .02em, padding .88/.94em,
  // border 2, radius 13, shadow 4/5; tail 10/7 at 42%/45%, 12 degrees.
  const caption = plan.text.find(t => t.id === 'caption');
  if (caption) {
    const unit = 3, font = 33, padX = font * .94, padY = font * .88, lineH = font * 1.55;
    ctx.font = `${font}px StudioDialogue`; ctx.letterSpacing = `${font * .02}px`;
    const lines = []; let line = '';
    for (const word of caption.text.split(' ')) {
      const next = line ? line + ' ' + word : word;
      if (line && ctx.measureText(next).width > 350) { lines.push(line); line = word; } else line = next;
    }
    lines.push(line);
    const width = Math.max(...lines.map(l => ctx.measureText(l).width)) + padX * 2 + 12;
    const height = lines.length * lineH + padY * 2 + 12;
    const skull = await stack('body', plan.layers.filter(l => l.slot === 'SK'));
    const skullX = skull ? bodyBox.x + ((skull.ink.x + skull.ink.width / 2) * body.canvas.width / skull.canvas.width - body.ink.x) * bodyScale
      : bodyBox.x + bodyBox.width / 2;
    const drawBubble = (ctx, x, y, facesRight) => {
      ctx.font = `${font}px StudioDialogue`; ctx.letterSpacing = `${font * .02}px`;
      ctx.save();
      ctx.translate(x + width / 2, y + height / 2);
      ctx.rotate((facesRight ? -2 : 2) * Math.PI / 180);
      ctx.translate(-x - width / 2, -y - height / 2);
      const rounded = (dx, dy, fill, stroke) => {
        const inset = stroke ? unit : 0;
        ctx.beginPath(); ctx.roundRect(x + dx + inset, y + dy + inset, width - inset * 2, height - inset * 2, 13 * unit - inset);
        ctx.fillStyle = fill; ctx.fill();
        if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = 2 * unit; ctx.stroke(); }
      };
      rounded(4 * unit, 5 * unit, 'rgba(0,0,0,.55)');
      rounded(0, 0, '#16151d', '#3b393d');
      const tail = (radius, inset, fraction, fill) => {
        const right = facesRight;
        ctx.save(); ctx.translate(right ? x + width - 6 + (inset - radius / 2) * unit : x + 6 - (inset - radius / 2) * unit, y + 6 + (height - 12) * fraction + radius * unit);
        ctx.scale(right ? 1 : -1, 1); ctx.rotate(12 * Math.PI / 180);
        ctx.beginPath(); ctx.moveTo(-radius * unit / 2, -radius * unit); ctx.lineTo(radius * unit / 2, 0); ctx.lineTo(-radius * unit / 2, radius * unit); ctx.closePath();
        ctx.fillStyle = fill; ctx.fill(); ctx.restore();
      };
      tail(10, 14, .42, '#3b393d'); tail(7, 8, .45, '#16151d');
      ctx.fillStyle = '#f2e9d7'; ctx.textBaseline = 'middle';
      lines.forEach((l, i) => ctx.fillText(l, x + 6 + padX, y + 6 + padY + lineH * (i + .5)));
      ctx.restore();
    };
    const preferred = { id: 'caption', x: plan.bubblePosition === 'left' ? 83 : 928 - width,
      y: 428, width: Math.ceil(width + 84), height: Math.ceil(height + 36) };
    const candidates = [];
    for (const facesRight of [true, false]) {
      const bubble = runtime.createCanvas(preferred.width, preferred.height);
      drawBubble(bubble.getContext('2d'), facesRight ? 12 : 57, 12, facesRight);
      const box = studioClearPosition(occupancy, preferred, [], { canvas: bubble,
        accept: x => (x + preferred.width / 2 < skullX) === facesRight });
      if (box) candidates.push({ box, bubble });
    }
    candidates.sort((a, b) => a.box.overlapPixels - b.box.overlapPixels
      || ((a.box.x - preferred.x) ** 2 + (a.box.y - preferred.y) ** 2)
        - ((b.box.x - preferred.x) ** 2 + (b.box.y - preferred.y) ** 2));
    if (!candidates.length) fail('No safe Studio bubble placement.');
    const { box: captionBox, bubble } = candidates[0];
    ctx.drawImage(bubble, captionBox.x, captionBox.y);
    information.push(captionBox);
    ctx.letterSpacing = '0px';
  }
  const markImg = await load('assets/brand/wordmark.png');
  const markCanvas = runtime.createCanvas(markImg.width, markImg.height);
  markCanvas.getContext('2d').drawImage(markImg, 0, 0);
  const mark = { canvas: markCanvas, ink: studioInk(markCanvas) };
  let markWidth = plan.markPosition === 'right' ? 340 : 480;
  let markHeight = markWidth * mark.ink.height / mark.ink.width;
  const preferredMark = { id: 'brand', x: plan.markPosition === 'right' ? 660 : 300,
    y: plan.markPosition === 'bottom' ? 1475 - markHeight : plan.markPosition === 'top' ? 280 : 800,
    width: markWidth, height: markHeight };
  const obstacles = [...information,
    ...plan.text.filter(t => t.id === 'friend-code').map(() => ({ x: 65, y: 1470, width: 600, height: 70 }))];
  let plate = studioClearPosition(occupancy, { ...preferredMark,
    width: markWidth + 32, height: markHeight + 36 }, obstacles);
  // The existing compact offering is equally valid when it can leave the art
  // clear. Keep the larger mark only when shrinking cannot buy clearance.
  if (plate.overlapPixels && markWidth > 340) {
    const compactHeight = 340 * mark.ink.height / mark.ink.width;
    const compact = studioClearPosition(occupancy, { ...preferredMark,
      width: 372, height: compactHeight + 36 }, obstacles);
    if (!compact.overlapPixels) { plate = compact; markWidth = 340; markHeight = compactHeight; }
  }
  const markBox = { ...preferredMark, x: plate.x + 12, y: plate.y + 12,
    width: markWidth, height: markHeight };
  ctx.fillStyle = '#2A2D28'; ctx.fillRect(plate.x + 8, plate.y + 10, plate.width - 8, plate.height - 10);
  ctx.fillStyle = '#F3EFE7'; ctx.fillRect(plate.x, plate.y, plate.width - 8, plate.height - 10);
  ctx.fillStyle = '#2A2D28'; ctx.fillRect(plate.x + 3, plate.y + 3, plate.width - 14, plate.height - 16);
  const before = ctx.getImageData(0, 0, 1080, 1920).data;
  paint(mark, markBox);
  const maskCanvas = runtime.createCanvas(1080, 1920), mc = maskCanvas.getContext('2d');
  mc.imageSmoothingEnabled = false;
  mc.drawImage(mark.canvas, mark.ink.x, mark.ink.y, mark.ink.width, mark.ink.height,
    markBox.x, markBox.y, markBox.width, markBox.height);
  const markContrast = studioMarkContrast(before, ctx.getImageData(0, 0, 1080, 1920).data,
    mc.getImageData(0, 0, 1080, 1920).data);
  assertStudioMarkContrast(markContrast);
  information.push(plate);
  for (const t of plan.text.filter(t => t.id === 'friend-code')) {
    ctx.font = `${t.size}px StudioBangers`; ctx.textBaseline = 'alphabetic';
    const m = ctx.measureText(t.text);
    const box = { id: t.id, x: t.x - m.actualBoundingBoxLeft, y: t.baseline - m.actualBoundingBoxAscent,
      width: m.actualBoundingBoxLeft + m.actualBoundingBoxRight, height: m.actualBoundingBoxAscent + m.actualBoundingBoxDescent };
    ctx.fillStyle = '#F3EFE7'; ctx.fillRect(box.x - 6, box.y - 6, box.width + 12, box.height + 12);
    ctx.fillStyle = '#2A2D28'; ctx.fillText(t.text, t.x, t.baseline);
    information.push({ ...box, x: box.x - 6, y: box.y - 6, width: box.width + 12, height: box.height + 12 });
  }
  assertStudioSafe([...bounds, ...information]);
  const blob = await runtime.encode(canvas);
  await assertStudioPng(blob);
  const renderLive = (target, stickers) => {
    const cx = target.getContext('2d'); cx.imageSmoothingEnabled = false;
    cx.clearRect(0, 0, 1080, 1920); cx.drawImage(baseCanvas, 0, 0);
    const liveBounds = bounds.filter(b => !b.id.startsWith('sticker-'));
    stickers.forEach((sticker, index) => {
      if (!figures[index]) return;
      const r = studioRasterSticker(figures[index], sticker, runtime), b = { id: `sticker-${index}`, ...r.box };
      cx.drawImage(r.decoration, b.x, b.y); cx.drawImage(r.art, b.x, b.y); liveBounds.push(b);
    });
    cx.drawImage(foreground, 0, 0);
    return liveBounds;
  };
  return { blob, plan, information, bounds, feetY, markContrast, renderLive, figures };
}

export async function assertStudioPng(blob) {
  if (!(blob instanceof Blob) || blob.type !== 'image/png' || blob.size < 45) fail('Studio requires a PNG image.');
  const h = new Uint8Array(await blob.slice(0, 24).arrayBuffer());
  const view = new DataView(h.buffer);
  if (![137,80,78,71,13,10,26,10].every((v, i) => h[i] === v)
    || String.fromCharCode(...h.slice(12, 16)) !== 'IHDR'
    || view.getUint32(16) !== 1080 || view.getUint32(20) !== 1920) fail('Studio requires a 1080x1920 PNG.');
}
